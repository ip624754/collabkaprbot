FILE: migrations/042_payments_fk_hardening.sql
================================================================================
-- 042: Payments FK hardening (never cascade-delete financial logs)
--
-- Problem:
--   stars_payments.user_id and payments.user_id were created with ON DELETE CASCADE.
--   If a user row is ever physically deleted, financial history disappears as well,
--   breaking auditability and potentially idempotency around charge IDs.
--
-- Fix:
--   Replace user_id FKs in payment ledgers with ON DELETE RESTRICT.
--   Keep all other user references (applying_by_user_id / applied_by_user_id) as SET NULL.

DO $$
DECLARE
  conname text;
  deltype "char";
BEGIN
  -- stars_payments.user_id
  IF EXISTS (SELECT 1 FROM information_schema.tables WHERE table_name='stars_payments') THEN
    SELECT c.conname, c.confdeltype
      INTO conname, deltype
    FROM pg_constraint c
    WHERE c.contype='f'
      AND c.conrelid='stars_payments'::regclass
      AND pg_get_constraintdef(c.oid) LIKE '%FOREIGN KEY (user_id)%REFERENCES users(id)%'
    LIMIT 1;

    IF conname IS NULL THEN
      EXECUTE 'ALTER TABLE stars_payments '
           || 'ADD CONSTRAINT stars_payments_user_id_fkey '
           || 'FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE RESTRICT';
    ELSIF deltype = 'c' THEN
      EXECUTE format('ALTER TABLE stars_payments DROP CONSTRAINT %I', conname);
      EXECUTE 'ALTER TABLE stars_payments '
           || 'ADD CONSTRAINT stars_payments_user_id_fkey '
           || 'FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE RESTRICT';
    END IF;
  END IF;

  -- payments.user_id
  conname := NULL;
  deltype := NULL;
  IF EXISTS (SELECT 1 FROM information_schema.tables WHERE table_name='payments') THEN
    SELECT c.conname, c.confdeltype
      INTO conname, deltype
    FROM pg_constraint c
    WHERE c.contype='f'
      AND c.conrelid='payments'::regclass
      AND pg_get_constraintdef(c.oid) LIKE '%FOREIGN KEY (user_id)%REFERENCES users(id)%'
    LIMIT 1;

    IF conname IS NULL THEN
      EXECUTE 'ALTER TABLE payments '
           || 'ADD CONSTRAINT payments_user_id_fkey '
           || 'FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE RESTRICT';
    ELSIF deltype = 'c' THEN
      EXECUTE format('ALTER TABLE payments DROP CONSTRAINT %I', conname);
      EXECUTE 'ALTER TABLE payments '
           || 'ADD CONSTRAINT payments_user_id_fkey '
           || 'FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE RESTRICT';
    END IF;
  END IF;
END$$;

-- Collabka migration pack: reconcile/repair (idempotent)
-- Purpose:
--   Bring ANY database to a minimum required infra state without conflicts.
--   Safe to run multiple times.
--
-- Notes:
--   - This script is NOT a full schema install. It only "patches" infra pieces that are safe to add.
--   - Everything here is written to be re-runnable (IF EXISTS / IF NOT EXISTS + guards).

-- 1) Crypto extension (sha256)
CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- 2) Outbox (reliable events)
CREATE TABLE IF NOT EXISTS event_outbox (
  id BIGSERIAL PRIMARY KEY,
  type TEXT NOT NULL,
  payload JSONB NOT NULL,
  status TEXT DEFAULT 'PENDING', -- PENDING, DONE, DEAD
  attempts INTEGER DEFAULT 0,
  next_attempt_at TIMESTAMPTZ,
  processed_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_outbox_worker
  ON event_outbox (status, next_attempt_at)
  WHERE status = 'PENDING';

-- 3) Audit (immutable events log)
CREATE TABLE IF NOT EXISTS audit_events (
  id BIGSERIAL PRIMARY KEY,
  tenant_id UUID,
  entity_type TEXT NOT NULL, -- 'giveaway', 'user', 'system'
  entity_id TEXT NOT NULL,
  event_type TEXT NOT NULL,
  actor_id TEXT,
  payload JSONB,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_audit_entity
  ON audit_events (entity_type, entity_id);

-- 4) Soft-delete support (safe add)
ALTER TABLE IF EXISTS barter_threads
  ADD COLUMN IF NOT EXISTS deleted_by_user_ids jsonb DEFAULT '[]'::jsonb;

ALTER TABLE IF EXISTS brand_leads
  ADD COLUMN IF NOT EXISTS deleted_by_user_ids jsonb DEFAULT '[]'::jsonb;

ALTER TABLE IF EXISTS brand_applications
  ADD COLUMN IF NOT EXISTS deleted_by_user_ids jsonb DEFAULT '[]'::jsonb;

# migration_pack — ручные операции с Postgres (Neon/restore/emergency)

Эта папка **не используется рантаймом** и **не должна** попадать в стандартный пайплайн миграций.
Она нужна как "страховка" для ручного переноса/восстановления Postgres (например, при миграции на новый Neon‑проект, когда схему/данные переносили dump/restore и таблица `schema_migrations` потерялась или стала неконсистентной).

## Когда это НЕ нужно
Если база новая/пустая — используйте только стандартный раннер миграций:

```bash
npm i
npm run migrate
```

## Основные сценарии

### A) Перенос/restore базы: схема и данные уже актуальные, но `schema_migrations` пустая/отсутствует
1) (Опционально, но полезно) привести БД к минимальному инфраструктурному состоянию (идемпотентно):

```bash
psql "$DATABASE_URL" -f migration_pack/01_reconcile.sql
```

2) **Только если вы уверены**, что схема реально соответствует текущему набору миграций в `migrations/`:

```bash
psql "$DATABASE_URL" -f migration_pack/00_mark_all_applied.sql
```

> ⚠️ Важно: `00_mark_all_applied.sql` **не выполняет миграции**, он лишь создаёт/заполняет `schema_migrations`.
> Используйте его только когда схема уже приведена в актуальный вид (например, вы переносили базу целиком).

3) Проверка (без применения):

```bash
node migrations/run.js --dry-run
```

Ожидаемо: все миграции будут `skip`.

### B) Быстрый repair после нестабильного переноса
Если после restore что-то "не хватает" из инфраструктуры (индексы/колонки/таблицы outbox/audit), можно безопасно повторить:

```bash
psql "$DATABASE_URL" -f migration_pack/01_reconcile.sql
```

## Что внутри
- `01_reconcile.sql` — безопасный (идемпотентный) "repair": включает минимальные инфраструктурные вещи (pgcrypto/outbox/audit/индексы/soft-delete/часть безопасных колонок и таблиц). Можно запускать повторно.
- `00_mark_all_applied.sql` — создаёт `schema_migrations` и отмечает **весь текущий** набор миграций как применённый (использовать осторожно).

## Важное про папку `migrations/`
В папке `migrations/` должны лежать **только настоящие миграции** вида `NNN_name.sql`.
Файлы `migration_pack/*` не должны дублироваться внутри `migrations/`, иначе раннер миграций может применить их как обычные миграции.
